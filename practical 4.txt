
# Distance between cities
distance = {
    'A': {'A': 0, 'B': 10, 'C': 15, 'D': 20},
    'B': {'A': 10, 'B': 0, 'C': 35, 'D': 25},
    'C': {'A': 15, 'B': 35, 'C': 0, 'D': 30},
    'D': {'A': 20, 'B': 25, 'C': 30, 'D': 0}
}

# Calculate total distance of a route
def total_distance(route):
    total = 0

    for i in range(len(route) - 1):
        total += distance[route[i]][route[i+1]]

    # Return to starting city
    total += distance[route[-1]][route[0]]

    return total


# Generate new routes by swapping cities
def get_neighbors(route):
    neighbors = []

    for i in range(1, len(route)):
        for j in range(i + 1, len(route)):

            new_route = list(route)

            new_route[i], new_route[j] = new_route[j], new_route[i]

            neighbors.append(tuple(new_route))

    return neighbors


# Local Beam Search
K = 3
iterations = 5

# Initial routes
beam = [
    ('A', 'B', 'C', 'D'),
    ('A', 'C', 'B', 'D'),
    ('A', 'D', 'B', 'C')
]

print("Initial Routes:")
for route in beam:
    print(route, "Distance =", total_distance(route))

# Repeat the search
for step in range(iterations):

    candidates = []

    # Generate neighboring routes
    for route in beam:
        candidates.append(route)
        candidates.extend(get_neighbors(route))

    # Remove duplicate routes
    unique_routes = []

    for route in candidates:
        if route not in unique_routes:
            unique_routes.append(route)

    # Sort routes by distance
    unique_routes.sort(key=total_distance)

    # Keep only K best routes
    beam = unique_routes[:K]

    print("\nIteration", step + 1)

    for route in beam:
        print(route, "Distance =", total_distance(route))


# Best route
best_route = beam[0]

print("\nBest Route:")
print(" -> ".join(best_route), "->", best_route[0])

print("Minimum Distance =", total_distance(best_route))
