
print("Chatbot: Hello! Welcome to SimpleBot.")
print("Chatbot: Type 'bye' to exit.")

while True:

    # Take input from user
    user_input = input("You: ").lower()

    # Pattern matching using predefined rules
    if "hello" in user_input or "hi" in user_input:
        print("Chatbot: Hello! How are you?")

    elif "how are you" in user_input:
        print("Chatbot: I am fine. Thank you!")

    elif "your name" in user_input:
        print("Chatbot: My name is SimpleBot.")

    elif "ai" in user_input:
        print("Chatbot: AI stands for Artificial Intelligence.")

    elif "help" in user_input:
        print("Chatbot: Sure! How can I help you?")

    elif "thank" in user_input:
        print("Chatbot: You are welcome!")

    elif "bye" in user_input:
        print("Chatbot: Goodbye! Have a nice day.")
        break

    else:
        print("Chatbot: Sorry, I don't understand.")
