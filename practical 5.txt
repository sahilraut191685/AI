

from collections import deque

start = (1, 2, 3,
         4, 0, 6,
         7, 5, 8)

goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)

queue = deque([(start, [])])
visited = {start}

while queue:

    state, path = queue.popleft()

    path = path + [state]

    if state == goal:
        print("Solution found!")
        
        for step in path:
            for i in range(0, 9, 3):
                print(step[i:i+3])
            print()
            
        break

    zero = state.index(0)

    moves = [-3, 3, -1, 1]

    for move in moves:

        new = zero + move

        if 0 <= new < 9:

            if move == -1 and zero % 3 == 0:
                continue

            if move == 1 and zero % 3 == 2:
                continue

            new_state = list(state)

            new_state[zero], new_state[new] = new_state[new], new_state[zero]

            new_state = tuple(new_state)

            if new_state not in visited:
                visited.add(new_state)
                queue.append((new_state, path))
