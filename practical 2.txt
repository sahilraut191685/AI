
import heapq

start = (1, 2, 3,
         4, 0, 6,
         7, 5, 8)

goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)


# Calculate Manhattan Distance
def heuristic(state):
    distance = 0

    for i in range(9):
        tile = state[i]

        if tile != 0:
            goal_pos = tile - 1

            distance += abs(i // 3 - goal_pos // 3)
            distance += abs(i % 3 - goal_pos % 3)

    return distance


# A* Search Algorithm
pq = [(heuristic(start), 0, start, [])]

visited = set()

while pq:

    f, g, state, path = heapq.heappop(pq)

    if state in visited:
        continue

    visited.add(state)

    path = path + [state]

    if state == goal:
        print("Solution Found!")
        print("Total Moves:", g)

        for s in path:
            for i in range(0, 9, 3):
                print(s[i:i+3])
            print()

        break

    zero = state.index(0)

    moves = [-3, 3, -1, 1]

    for move in moves:
        new = zero + move

        if 0 <= new < 9:
            if not (move == -1 and zero % 3 == 0) and \
               not (move == 1 and zero % 3 == 2):

                new_state = list(state)

                new_state[zero], new_state[new] = \
                    new_state[new], new_state[zero]

                new_state = tuple(new_state)

                if new_state not in visited:
                    new_g = g + 1
                    new_h = heuristic(new_state)
                    new_f = new_g + new_h

                    heapq.heappush(
                        pq,
                        (new_f, new_g, new_state, path)
                    )
else:
    print("No solution")

Practical 2: 8 Puzzle Problem using Uniform Cost Search
Aim: To implement the 8 Puzzle Problem using Uniform Cost Search in Python.

Theory:

The 8 Puzzle Problem is a classical Artificial Intelligence search problem consisting of a 3 × 3 board containing eight numbered tiles and one blank space represented by zero.

The objective is to rearrange the tiles from the initial state to the desired goal state by moving adjacent tiles into the blank space.

Uniform Cost Search (UCS) is an uninformed search algorithm that expands the node having the lowest cumulative path cost. It uses a priority queue to select the next node for exploration.

In this algorithm, each movement has a cost of 1. The algorithm starts from the initial state, generates possible successor states, and inserts them into the priority queue according to their path costs.

A visited set is maintained to avoid exploring the same state repeatedly. The process continues until the goal state is reached or no states remain to explore.

The evaluation function is:
\[
f(n)=g(n)
\]

Where:
- \(g(n)\) is the path cost from the initial state to the current node.
- \(f(n)\) is the priority of the node.
Conclusion: The 8 Puzzle problem is solved using Uniform Cost Search by exploring states according to their cumulative path costs.
